메모장에 바로 복사해서 사용할 수 있도록 HTML, CSS, 자바스크립트가 모두 포함된 최종 코드를 만들어 드립니다. 이 코드를 메모장에 붙여넣고 index.html로 저장한 뒤 GitHub에 올리시면 됩니다.

사용자가 금액을 입력하고 버튼을 누르면 설정한 계좌로 연결되는 기능을 담았습니다.

💻 웹사이트 최종 코드 (index.html)

    <div class="container">
        <h1>💸 송금하기</h1>
        <p>보낼 금액을 입력하세요</p>
        <input type="number" id="amount" placeholder="금액 입력" value="1000">
        <button onclick="sendMoney()">토스로 송금하기</button>
    </div>

    <!-- JavaScript: 기능 작동 -->
    <script>
        function sendMoney() {
            // 1. 입력한 금액 가져오기
            const amount = document.getElementById('amount').value;
            
            // 2. 내 정보 설정 (내 토스 아이디 혹은 계좌 정보)
            // 토스아이디가 없다면 주소를 'supertoss://send?amount=' + amount + '&accountNo=계좌번호&bank=은행명' 형태로 수정 가능
            const tossId = "내아이디"; 
            const url = `https://toss.im/_m/${tossId}?amount=${amount}`;

            // 3. 페이지 이동
            if (amount > 0) {
                location.href = url;
            } else {
                alert("금액을 정확히 입력해주세요!");
            }
        }
    </script>

</body>
</html>
